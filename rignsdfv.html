self.avail_w = w inner_w = w - self.pad_left - self.pad_right if self.border_left_color: inner_w -= 4 # border width total_h = self.pad_top + self.pad_bottom self._heights = [] for p in self.content_paras: pw, ph = p.wrap(inner_w, availHeight) self._heights.append(ph) total_h += ph if len(self.content_paras) > 1: total_h += (len(self.content_paras) - 1) * 4 self._h = total_h return w, total_h def draw(self): w = self.avail_w h = self._h c = self.canv # background c.setFillColor(self.bg) c.roundRect(0, 0, w, h, 4, fill=1, stroke=0) # left border if self.border_left_color: c.setFillColor(self.border_left_color) c.rect(0, 0, 3, h, fill=1, stroke=0) # draw content x = self.pad_left + (4 if self.border_left_color else 0) y = h - self.pad_top for i, p in
